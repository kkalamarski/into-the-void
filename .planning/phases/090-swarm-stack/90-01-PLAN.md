Create `docker-stack.yml` at project root with all services configured for Docker Swarm mode.

**File structure:**
```yaml
version: '3.8'

services:
  # Database services
  postgres:
    image: postgres:16-alpine
    deploy:
      placement:
        constraints:
          - node.role == manager  # Database on manager for volume access
      resources:
        limits:
          cpus: '1.0'
          memory: 512M
        reservations:
          cpus: '0.25'
          memory: 256M
    environment:
      POSTGRES_USER: ${POSTGRES_USER:-postgres}
      POSTGRES_PASSWORD: ${POSTGRES_PASSWORD}
      POSTGRES_DB: ${POSTGRES_DB:-into_the_void}
    volumes:
      - postgres_data:/var/lib/postgresql/data
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U postgres"]
      interval: 10s
      timeout: 5s
      retries: 5
    networks:
      - internal

  redis:
    image: redis:7-alpine
    deploy:
      placement:
        constraints:
          - node.role == manager
      resources:
        limits:
          cpus: '0.5'
          memory: 256M
        reservations:
          cpus: '0.1'
          memory: 128M
    command: redis-server --appendonly yes
    volumes:
      - redis_data:/data
    healthcheck:
      test: ["CMD", "redis-cli", "ping"]
      interval: 10s
      timeout: 5s
      retries: 5
    networks:
      - internal

  # Application services
  api:
    image: ${REGISTRY:-ghcr.io/owner/repo}/api:${TAG:-latest}
    deploy:
      replicas: 1
      update_config:
        parallelism: 1
        delay: 10s
        failure_action: rollback
      resources:
        limits:
          cpus: '1.0'
          memory: 512M
        reservations:
          cpus: '0.25'
          memory: 256M
    environment:
      NODE_ENV: production
      DATABASE_URL: postgresql://${POSTGRES_USER:-postgres}:${POSTGRES_PASSWORD}@postgres:5432/${POSTGRES_DB:-into_the_void}
      JWT_SECRET: ${JWT_SECRET}
      REDIS_HOST: redis
      REDIS_PORT: 6379
    depends_on:
      postgres:
        condition: service_healthy
      redis:
        condition: service_healthy
    networks:
      - internal
      - traefik-public
    labels:
      - "traefik.enable=true"
      - "traefik.http.routers.api.rule=Host(`api.${DOMAIN:-intothevoid.online}`)"
      - "traefik.http.services.api.loadbalancer.server.port=3000"

  game-server:
    image: ${REGISTRY:-ghcr.io/owner/repo}/game-server:${TAG:-latest}
    deploy:
      replicas: 1
      update_config:
        parallelism: 1
        delay: 10s
        failure_action: rollback
      resources:
        limits:
          cpus: '2.0'
          memory: 1024M
        reservations:
          cpus: '0.5'
          memory: 512M
    environment:
      NODE_ENV: production
      DATABASE_URL: postgresql://${POSTGRES_USER:-postgres}:${POSTGRES_PASSWORD}@postgres:5432/${POSTGRES_DB:-into_the_void}
      JWT_SECRET: ${JWT_SECRET}
      REDIS_HOST: redis
      REDIS_PORT: 6379
    depends_on:
      postgres:
        condition: service_healthy
      redis:
        condition: service_healthy
    networks:
      - internal
      - traefik-public
    labels:
      - "traefik.enable=true"
      - "traefik.http.routers.game-server.rule=Host(`game.${DOMAIN:-intothevoid.online}`)"
      - "traefik.http.services.game-server.loadbalancer.server.port=3001"
      - "traefik.http.services.game-server.loadbalancer.sticky.cookie=true"
      - "traefik.http.services.game-server.loadbalancer.sticky.cookie.name=game_session"

  web:
    image: ${REGISTRY:-ghcr.io/owner/repo}/web:${TAG:-latest}
    deploy:
      replicas: 1
      update_config:
        parallelism: 1
        delay: 10s
        failure_action: rollback
      resources:
        limits:
          cpus: '0.5'
          memory: 128M
        reservations:
          cpus: '0.1'
          memory: 64M
    networks:
      - traefik-public
    labels:
      - "traefik.enable=true"
      - "traefik.http.routers.web.rule=Host(`play.${DOMAIN:-intothevoid.online}`)"
      - "traefik.http.services.web.loadbalancer.server.port=80"

volumes:
  postgres_data:
    driver: local
  redis_data:
    driver: local

networks:
  internal:
    driver: overlay
    internal: true  # No external access
  traefik-public:
    external: true  # Created by Traefik stack in Phase 91
```

**Important notes:**
- Use `${VAR:-default}` syntax for environment variable substitution
- Database password MUST NOT have default (requires explicit setting)
- Services use overlay networks for Swarm communication
- `internal` network has no external access (db isolation)
- `traefik-public` is external network created in Phase 91
- Traefik labels added now but router won't work until Phase 91
- Game server gets higher resources (real-time processing)
- Web service gets minimal resources (static file serving)

```env
POSTGRES_DB=into_the_void

# Container registry (GHCR by default)

# Image tag (defaults to latest)
TAG=latest

# ================================
# Deployment Notes
# ================================
#
# 1. Generate secure passwords:
#    openssl rand -base64 32
#
# 2. Update REGISTRY to match your GitHub Container Registry:
#    REGISTRY=ghcr.io/your-username/into-the-void
#
# 3. Deploy with:
#    docker stack deploy -c docker-stack.yml --with-registry-auth itv
#
# 4. Check status:
#    docker stack services itv
#    docker service logs itv_api
```
