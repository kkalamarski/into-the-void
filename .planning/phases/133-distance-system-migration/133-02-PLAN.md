```typescript
private async getNearbyCreatures(
  zoneId: string,
  centerX: number,
  centerY: number,
  radius: number,
  excludeId?: string,
): Promise<Creature[]> {
  const entities = await this.zonesService.getZoneEntities(zoneId);
  // Convert tile center to pixel coordinates for distance comparison
  const { px: cPx, py: cPy } = tileToPixelCenter(centerX, centerY);
  const radiusPx = radius * TILE_SIZE_PX;
  return entities.filter(
    (e): e is Creature =>
      e.type === 'creature' &&
      (e as Creature).active &&
      (e as Creature).health > 0 &&
      e.id !== excludeId &&
      (() => {
        const { px: ePx, py: ePy } = tileToPixelCenter(e.position.x, e.position.y);
        return pixelDistanceTo(cPx, cPy, ePx, ePy) <= radiusPx;
      })(),
  );
}
```
Alternatively, compute the pixel distance inline without the IIFE for readability — extract creature coords in a simple map-filter pattern.

```typescript
/**
 * Cancel active gather if player moved beyond GATHER_RANGE_PX.
 * Called from MovementService tick path (Phase 133).
 * Per user decision: immediate cancel, full progress reset, no toast.
 */
cancelIfOutOfRange(playerId: string, playerPx: number, playerPy: number): void {
  const active = this.activeChallenges.get(playerId);
  if (!active) return;

  const { px: ex, py: ey } = tileToPixelCenter(active.entity.position.x, active.entity.position.y);
  const dist = pixelDistanceTo(playerPx, playerPy, ex, ey);

  if (dist > GATHER_RANGE_PX) {
    this.entityLocks.delete(active.entityId);
    this.activeChallenges.delete(playerId);
    // No toast or "Too far away" message — progress bar simply disappears (per user decision)
  }
}
```
