### Task 2: Implement handleGatherEffect method

**File**: `apps/game-server/src/game/ability.service.ts`

Add new method:

```typescript
private async handleGatherEffect(
  playerId: string,
  targetEntityId: string,
  effect: GatherEffect,
  toolStats: { yieldBonus?: number; gatherSpeed?: number }
): Promise<{ success: boolean; error?: string }> {
  // 1. Get target entity
  const entity = this.entityService.getEntity(targetEntityId);
  if (!entity) {
    return { success: false, error: 'Target not found' };
  }

  // 2. Validate entity type matches gather type
  if (effect.gatherType === 'harvest' && entity.type !== 'plant') {
    return { success: false, error: 'Cannot harvest this target' };
  }
  if (effect.gatherType === 'mine' && entity.type !== 'mineral') {
    return { success: false, error: 'Cannot mine this target' };
  }

  // 3. Calculate yield with tool bonus
  const yieldMultiplier = 1 + (toolStats.yieldBonus ?? 0);
  const finalYield = Math.floor(effect.baseYield * yieldMultiplier);

  // 4. Call EntityService to process gathering
  const result = await this.entityService.handleToolUse(
    playerId,
    targetEntityId,
    finalYield
  );

  if (!result.success) {
    return { success: false, error: result.error };
  }

  // 5. Emit events for entity update and loot
  // (EntityService handles inventory updates internally)

  return { success: true };
}
```

### Task 5: Remove/simplify gathering:start handler

**File**: `apps/game-server/src/game/game.gateway.ts`

The `gathering:start` handler can be removed or simplified since gathering now goes through ability:use. If keeping for backward compatibility, redirect to ability system:

```typescript
@SubscribeMessage('gathering:start')
async handleGatheringStart(
  @ConnectedSocket() client: Socket,
  @MessageBody() data: { targetEntityId: string }
): Promise<void> {
  // Redirect to ability system
  // Determine ability based on entity type
  const entity = this.entityService.getEntity(data.targetEntityId);
  if (!entity) {
    client.emit('error', { code: 'GATHERING_ERROR', message: 'Target not found' });
    return;
  }

  const abilityId = entity.type === 'plant' ? 'harvest' : 'mine';

  // Delegate to ability:use
  await this.handleAbilityUse(client, {
    abilityId,
    targetEntityId: data.targetEntityId
  });
}
```
