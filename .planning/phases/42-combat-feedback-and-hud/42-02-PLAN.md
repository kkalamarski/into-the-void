  <action>
Create a new Zustand store `combatStore.ts` in the store directory:

```typescript
import { create } from 'zustand';
import { gameSocket } from '../network/socket';
import { useGameStore } from './gameStore';

interface CombatState {
  inCombat: boolean;
  targetEntityId: string | null;
  setInCombat: (inCombat: boolean, targetEntityId?: string | null) => void;
}

export const useCombatStore = create<CombatState>((set) => ({
  inCombat: false,
  targetEntityId: null,
  setInCombat: (inCombat, targetEntityId = null) => set({ inCombat, targetEntityId }),
}));

// Listen for combat:start - player entered combat
// Two payload shapes:
// 1. Player-initiated (from game.gateway.ts): { active: true, turn, participants[], currentActorId, startedAt }
// 2. Creature aggro (from combat.service.ts): { attackerId: creatureId, defenderId: playerId, timestamp }
gameSocket.on('combat:start', (data: {
  // Creature aggro fields
  attackerId?: string;
  defenderId?: string;
  timestamp?: number;
  // Player-initiated fields
  active?: boolean;
  currentActorId?: string;
}) => {
  const currentPlayer = useGameStore.getState().player;
  if (!currentPlayer) return;

  // Check if this combat involves the local player
  const isPlayerInvolved =
    data.currentActorId === currentPlayer.id || // Player initiated attack
    data.defenderId === currentPlayer.id;       // Creature attacked player

  if (isPlayerInvolved) {
    // For player-initiated: no target ID in payload, set null (will be updated on combat:damage)
    // For creature aggro: attacker is the creature we're fighting
    useCombatStore.getState().setInCombat(true, data.attackerId ?? null);
  }
});

// Listen for player:death - combat ends when player dies
gameSocket.on('player:death', (data: { playerId: string }) => {
  const currentPlayer = useGameStore.getState().player;
  if (currentPlayer && data.playerId === currentPlayer.id) {
    useCombatStore.getState().setInCombat(false);
  }
});

// Listen for entity:update with killed creature (active: false on combat target)
// This handles: creature killed, creature leashed (returns to spawn and disengages)
// Note: We could also use combat:damage with killed: true, but entity:update is more reliable
// for catching all combat-ending scenarios including leash
gameSocket.on('entity:update', (data: { entityId: string; changes: { active?: boolean } }) => {
  const { targetEntityId, inCombat } = useCombatStore.getState();

  // If the entity we're fighting becomes inactive, combat ends
  if (inCombat && targetEntityId === data.entityId && data.changes.active === false) {
    useCombatStore.getState().setInCombat(false);
  }
});

// Also handle combat:damage with killed: true for immediate feedback
gameSocket.on('combat:damage', (data: { attackerId: string; defenderId: string; killed: boolean }) => {
  const currentPlayer = useGameStore.getState().player;
  const { targetEntityId, inCombat } = useCombatStore.getState();

  // If we're in combat but don't have targetEntityId set (player-initiated case),
  // set it from the first damage event where we're the attacker
  if (inCombat && !targetEntityId && currentPlayer && data.attackerId === currentPlayer.id) {
    useCombatStore.getState().setInCombat(true, data.defenderId);
  }

  // If we killed our target, combat ends
  const currentTarget = useCombatStore.getState().targetEntityId;
  if (inCombat && currentTarget === data.defenderId && data.killed) {
    useCombatStore.getState().setInCombat(false);
  }
});
```

This store:
1. Tracks `inCombat` boolean state
2. Stores `targetEntityId` to know which entity we're fighting
3. Sets combat=true on combat:start (player-initiated or creature aggro)
4. Sets combat=false when: player dies, target creature dies (killed flag), or target becomes inactive (leash/despawn)
