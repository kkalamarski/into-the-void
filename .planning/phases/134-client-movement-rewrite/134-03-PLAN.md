**ConnectionQualityMonitor (`apps/web/src/game/systems/ConnectionQualityMonitor.ts`):**

A simple class that tracks connection quality based on positionCorrection frequency.

```typescript
export type ConnectionQuality = 'good' | 'degraded' | 'poor';

export class ConnectionQualityMonitor {
  private correctionTimestamps: number[] = [];
  private readonly WINDOW_MS = 5000; // 5-second sliding window
  private readonly DEGRADED_THRESHOLD = 3; // >3 corrections in 5s = degraded
  private readonly POOR_THRESHOLD = 8; // >8 corrections in 5s = poor
  private currentQuality: ConnectionQuality = 'good';
  private onQualityChange: ((quality: ConnectionQuality) => void) | null = null;

  setOnQualityChange(callback: (quality: ConnectionQuality) => void): void {
    this.onQualityChange = callback;
  }

  recordCorrection(): void {
    const now = Date.now();
    this.correctionTimestamps.push(now);
    // Prune old entries
    this.correctionTimestamps = this.correctionTimestamps.filter(t => now - t < this.WINDOW_MS);
    this.evaluate();
  }

  private evaluate(): void {
    const count = this.correctionTimestamps.length;
    let newQuality: ConnectionQuality;
    if (count > this.POOR_THRESHOLD) {
      newQuality = 'poor';
    } else if (count > this.DEGRADED_THRESHOLD) {
      newQuality = 'degraded';
    } else {
      newQuality = 'good';
    }
    if (newQuality !== this.currentQuality) {
      this.currentQuality = newQuality;
      this.onQualityChange?.(newQuality);
    }
  }

  getQuality(): ConnectionQuality {
    // Prune before reading
    const now = Date.now();
    this.correctionTimestamps = this.correctionTimestamps.filter(t => now - t < this.WINDOW_MS);
    this.evaluate();
    return this.currentQuality;
  }

  reset(): void {
    this.correctionTimestamps = [];
    this.currentQuality = 'good';
    this.onQualityChange?.('good');
  }
}
```

**ConnectionIndicator (`apps/web/src/components/hud/ConnectionIndicator.tsx`):**

A small React component showing a colored dot in the corner.

```tsx
import { useGameStore } from '../../store/gameStore';
import './ConnectionIndicator.css';

export function ConnectionIndicator() {
  const connectionQuality = useGameStore((s) => s.connectionQuality);

  // Only show when NOT good (subtle — disappears when connection is fine)
  if (connectionQuality === 'good') return null;

  const color = connectionQuality === 'degraded' ? 'var(--color-warning, #ffaa00)' : 'var(--color-danger, #ff4444)';
  const label = connectionQuality === 'degraded' ? 'Unstable connection' : 'Poor connection';

  return (
    <div className="connection-indicator" title={label}>
      <div className="connection-dot" style={{ backgroundColor: color }} />
    </div>
  );
}
```
