```typescript
static createFloatingError(scene: Phaser.Scene, x: number, y: number, message: string): void {
  const startY = y - 128;
  const text = scene.add.text(x, startY, message, {
    fontSize: '48px',
    fontStyle: 'bold',
    color: '#ff4444',  // Red for errors
    stroke: '#000000',
    strokeThickness: 4,
    shadow: {
      offsetX: 2,
      offsetY: 2,
      color: '#000000',
      blur: 4,
      fill: true,
    },
  });
  text.setOrigin(0.5, 0.5);
  text.setDepth(99999);

  scene.tweens.add({
    targets: text,
    y: startY - 80,
    alpha: 0,
    duration: 1200,
    ease: 'Cubic.easeOut',
    onComplete: () => { text.destroy(); },
  });
}
```

Use 48px font (smaller than damage 64px) so error text doesn't overpower damage numbers. Duration 1200ms (slightly longer than damage at 1000ms) so errors are readable.

**abilityStore.ts** — Replace the `ability:result` handler (lines 134-146) with:

```typescript
gameSocket.on('ability:result', (data) => {
  if (data.success && data.cooldownEndsAt) {
    useAbilityStore.getState().setCooldown(data.abilityId, data.cooldownEndsAt);
  }
  if (data.success) {
    useAbilityStore.getState().clearCast();
  }

  // Show floating error text for failed abilities (ABIL-03)
  if (!data.success && data.error) {
    const worldScene = (window as any).__phaserWorldScene;
    if (worldScene?.showErrorText) {
      worldScene.showErrorText(data.error);
    }
  }

  // Record successful damage in combat log (ABIL-01)
  if (data.success && data.damage && data.damage > 0) {
    const { useCombatLogStore } = require('./combatLogStore');
    // Get target name from entity store
    const { useEntityStore } = require('./entityStore');
    const { useCombatStore } = require('./combatStore');
    const targetId = useCombatStore.getState().selectedTarget;
    let targetName = 'Unknown';
    if (targetId) {
      const entity = useEntityStore.getState().entities.get(targetId);
      if (entity && 'name' in entity) {
        targetName = (entity as any).name ?? entity.type;
      } else if (entity) {
        targetName = entity.type;
      }
    }
    useCombatLogStore.getState().addEntry({
      timestamp: Date.now(),
      type: 'dealt',
      damage: data.damage,
      targetName,
      critical: false,
      killed: data.targetHealth !== undefined && data.targetHealth <= 0,
    });
  }
});
```
