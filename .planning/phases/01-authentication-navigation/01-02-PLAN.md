```typescript
import { apiCall } from '../utils/api';
import { useAuthStore } from '../store/authStore';
import { useNavigate } from 'react-router';

// In component:
const { login } = useAuthStore();
const navigate = useNavigate();

const handleSubmit = async (e: React.FormEvent<HTMLFormElement>) => {
  e.preventDefault();
  setLoading(true);
  setError(null);

  const formData = new FormData(e.currentTarget);
  const email = formData.get('email') as string;
  const password = formData.get('password') as string;

  try {
    const response = await apiCall<{ token: string; user: { accountId: string; email: string } }>(
      '/auth/login',
      {
        method: 'POST',
        body: JSON.stringify({ email, password }),
      }
    );
    login(response.token, response.user);
    navigate('/character-select');
  } catch (err) {
    setError('Invalid email or password');
  } finally {
    setLoading(false);
  }
};
```

**RegisterScreen.tsx:**
- Form with email, password, confirmPassword fields
- Client-side validation: passwords must match (show error inline)
- Use native HTML5 validation (required, type="email", minLength=8)
- useState for loading, error
- On submit: validate passwords match, call apiCall POST /auth/register with {email, password}
- On success: call login(token, user) from useAuthStore, navigate to /character-select
- On error: display error message from API (e.g., "Email already registered")
- Disable form during loading
- Link to login screen at bottom

Import and use apiCall like this:
```typescript
import { apiCall } from '../utils/api';
import { useAuthStore } from '../store/authStore';
import { useNavigate } from 'react-router';

// In component:
const { login } = useAuthStore();
const navigate = useNavigate();

const handleSubmit = async (e: React.FormEvent<HTMLFormElement>) => {
  e.preventDefault();

  const formData = new FormData(e.currentTarget);
  const email = formData.get('email') as string;
  const password = formData.get('password') as string;
  const confirmPassword = formData.get('confirmPassword') as string;

  if (password !== confirmPassword) {
    setError('Passwords do not match');
    return;
  }

  setLoading(true);
  setError(null);

  try {
    const response = await apiCall<{ token: string; user: { accountId: string; email: string } }>(
      '/auth/register',
      {
        method: 'POST',
        body: JSON.stringify({ email, password }),
      }
    );
    login(response.token, response.user);
    navigate('/character-select');
  } catch (err) {
    setError(err instanceof Error ? err.message : 'Registration failed');
  } finally {
    setLoading(false);
  }
};
```
