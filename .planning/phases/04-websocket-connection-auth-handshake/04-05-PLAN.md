Create GameScreen.tsx that orchestrates the full connection flow:

1. Imports:
   - React, useEffect, useState
   - useNavigate from react-router
   - useAuthStore, useCharacterStore, useGameStore
   - gameSocket from network/socket
   - LoadingScreen, ErrorModal from components
   - GameContainer from components
   - getErrorInfo from @into-the-void/shared-types

2. State:
   - error: ErrorCodeInfo | null for showing ErrorModal

3. Get from stores:
   - token from authStore
   - selectedCharacterId from characterStore
   - loadingStage, setLoadingStage, setLoadingProgress, setPlayer from gameStore

4. Connection flow in useEffect (runs once on mount):
   ```typescript
   useEffect(() => {
     const connect = async () => {
       // Validate we have auth data
       if (!token || !selectedCharacterId) {
         navigate('/character-select');
         return;
       }

       try {
         // Stage 1: Connecting (0-20%)
         setLoadingStage('connecting');
         setLoadingProgress(10);

         const serverUrl = import.meta.env.VITE_GAME_SERVER_URL || 'http://localhost:3001';
         gameSocket.connect(serverUrl);

         setLoadingProgress(20);

         // Stage 2: Authenticating (20-40%)
         setLoadingStage('authenticating');
         setLoadingProgress(30);

         const player = await gameSocket.authenticate(token, selectedCharacterId);
         setPlayer(player);
         setLoadingProgress(40);

         // Stage 3: Loading world (40-90%)
         setLoadingStage('loading-world');
         // zone:state event will be received - wait for it
         // This is handled by socket event listener
         setLoadingProgress(70);

         // Stage 4: Spawning (90-100%)
         setLoadingStage('spawning');
         setLoadingProgress(90);

         // Brief delay for visual completion
         await new Promise(resolve => setTimeout(resolve, 500));

         setLoadingProgress(100);
         setLoadingStage('ready');

       } catch (err) {
         const message = err instanceof Error ? err.message : 'Connection failed';

         // Map error to ErrorCodeInfo
         if (message.includes('timeout')) {
           setError(getErrorInfo('AUTH_TIMEOUT'));
         } else if (message.includes('token') || message.includes('Authentication')) {
           setError(getErrorInfo('AUTH_FAILED'));
         } else if (message.includes('Character')) {
           setError(getErrorInfo('INVALID_CHARACTER'));
         } else {
           setError(getErrorInfo('SERVER_ERROR'));
         }
       }
     };

     connect();

     return () => {
       // Cleanup on unmount
       gameSocket.disconnect();
       setLoadingStage('idle');
     };
   }, [token, selectedCharacterId]);
   ```

5. Handle error actions:
   ```typescript
   const handleErrorAction = () => {
     if (!error) return;

     switch (error.action) {
       case 'redirect-login':
         navigate('/login', { state: { error: error.message } });
         break;
       case 'redirect-characters':
         navigate('/character-select', { state: { error: error.message } });
         break;
       case 'retry':
         setError(null);
         window.location.reload(); // Simple retry
         break;
       default:
         setError(null);
     }
   };
   ```

6. Render:
   ```tsx
   return (
     <>
       {error && (
         <ErrorModal error={error} onRetry={handleErrorAction} />
       )}

       {loadingStage !== 'ready' && loadingStage !== 'idle' && !error && (
         <LoadingScreen />
       )}

       {loadingStage === 'ready' && (
         <GameContainer />
       )}
     </>
   );
   ```
