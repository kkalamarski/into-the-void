   Add a `applyFrenzyEffect()` private method:
   ```typescript
   /**
    * CRAI-06: Apply or remove frenzy red tint overlay on a creature.
    * Red tint (0xff4444) with pulsing glow that syncs with faster attack speed.
    */
   private applyFrenzyEffect(entityId: string, frenzied: boolean): void {
     const container = this.entitySprites.get(entityId);
     if (!container) return;

     const sprite = container.getAt(0) as Phaser.GameObjects.Sprite;
     if (!sprite) return;

     if (frenzied) {
       // Apply red tint
       sprite.setTint(0xff4444);

       // Add pulsing glow tween (syncs with 2x attack speed feel)
       if (!container.getData('frenzyTween')) {
         const tween = this.scene.tweens.add({
           targets: sprite,
           alpha: { from: 1.0, to: 0.6 },
           duration: 400,  // Fast pulse matching doubled attack speed
           yoyo: true,
           repeat: -1,     // Infinite loop
           ease: 'Sine.easeInOut',
         });
         container.setData('frenzyTween', tween);
       }
     } else {
       // Remove frenzy effects
       sprite.clearTint();
       sprite.setAlpha(1.0);

       const tween = container.getData('frenzyTween') as Phaser.Tweens.Tween | null;
       if (tween) {
         tween.destroy();
         container.setData('frenzyTween', null);
       }
     }
   }
   ```
