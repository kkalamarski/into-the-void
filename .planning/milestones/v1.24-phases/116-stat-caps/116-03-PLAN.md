function CharStatRow({ statKey, label, base, equipment, total, raw }: CharStatRowProps) {
  const [showDRTooltip, setShowDRTooltip] = React.useState(false);
  const Icon = STAT_ICONS[statKey] ?? GiHearts;
  const isDR = raw > 200;
  const isCapped = total >= 400;

  // Build CSS class list
  const rowClasses = [
    'char-stat-row',
    'char-stat--has-threshold',  // always show subtle 200 marker
    isCapped ? 'char-stat--capped' : isDR ? 'char-stat--dr' : '',
  ].filter(Boolean).join(' ');

  return (
    <div
      className={rowClasses}
      onMouseEnter={() => isDR && setShowDRTooltip(true)}
      onMouseLeave={() => setShowDRTooltip(false)}
    >
      <Icon className="char-stat-icon" />
      <span className="char-stat-label">{label}</span>
      <span className="char-stat-total">{total}</span>
      {isCapped && <span className="char-stat-cap-label">CAPPED</span>}
      <span className="char-stat-breakdown">
        {equipment !== 0 ? `(${base}+${equipment})` : ''}
      </span>
      {showDRTooltip && isDR && (
        <div className="char-stat-dr-tooltip">
          Raw: {raw} | Effective: {total}
        </div>
      )}
    </div>
  );
}

      // Check if this stat is in DR territory for the current character
      const currentRaw = currentStats?.raw?.[stat] ?? 0;
      const isInDR = currentRaw > 200;

      // Compute effective delta if in DR territory
      let effectiveDelta = delta;
      let showDR = false;
      if (isInDR && delta !== 0) {
        // Effective delta = DR(currentRaw + delta) - DR(currentRaw)
        const currentEffective = applyDiminishingReturns(currentRaw);
        const newEffective = applyDiminishingReturns(currentRaw + delta);
        effectiveDelta = newEffective - currentEffective;
        // Only show DR annotation if effective differs from raw delta
        showDR = effectiveDelta !== delta;
      }

      return (
        <div
          key={stat}
          className={`tooltip-delta ${effectiveDelta > 0 ? 'tooltip-delta--positive' : effectiveDelta < 0 ? 'tooltip-delta--negative' : 'tooltip-delta--neutral'}`}
        >
